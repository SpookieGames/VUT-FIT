\subsection*{1. Zostavenie diferenciálnej rovnice}
Vychádzame zo základných fyzikálnych vzťahov (Ohmov zákon a II. Kirchhoffov zákon):

\begin{enumerate}
    \item $i = \frac{u_R}{R}$ (Ohmov zákon)
    \item $u_R + u_C - U = 0 \implies u_R = U - u_C$ (II. Kirchhoffov zákon)
    \item $u_C' = \frac{i}{C} \implies i = C \cdot u_C'$ (Definícia prúdu kondenzátorom)
\end{enumerate}

Dosadením vzťahu (1) a (3) do rovnice (2) vyjadríme napätie na rezistore pomocou derivácie napätia na kondenzátore:
$$ R \cdot C \cdot u_C' + u_C = U $$

Vydelením rovnice súčinom $R \cdot C$ dostaneme nehomogénnu lineárnu diferenciálnu rovnicu 1. rádu (D.R.):
\begin{equation}
    u_C' + \frac{1}{R \cdot C} u_C = \frac{U}{R \cdot C}, \quad u_C(0) = u_{CP}
    \label{eq:diff_obecna}
\end{equation}

Dosadenie hodnôt ($R \cdot C = 100 \cdot 2 = 200$):
\begin{equation}
    u_C' + \frac{1}{200} u_C = \frac{28}{200} \implies u_C' + \frac{1}{200} u_C = \frac{7}{50}
    \label{eq:diff_num}
\end{equation}

\subsection*{2. Analytické riešenie metódou variácie konštanty}

\textbf{A) Riešenie homogénnej časti:}
Položíme pravú stranu rovnú nule:
$$ u_C' + \frac{1}{R \cdot C} u_C = 0 $$
Charakteristická rovnica:
$$ \lambda + \frac{1}{R \cdot C} = 0 \implies \lambda = -\frac{1}{R \cdot C} = -\frac{1}{200} $$
Tvar homogénneho riešenia (kde $K$ nahradíme funkciou času $K(t)$ pre variáciu):
\begin{equation}
    u_C(t) = K(t) \cdot e^{\lambda t} = K(t) \cdot e^{-\frac{t}{200}}
\end{equation}

\textbf{B) Variácia konštanty:}
Zderivujeme predpokladané riešenie $u_C(t)$ podľa pravidla o derivácii súčinu:
$$ u_C'(t) = K'(t) \cdot e^{-\frac{t}{200}} + K(t) \cdot \left( -\frac{1}{200} \right) \cdot e^{-\frac{t}{200}} $$

Dosadíme $u_C(t)$ a $u_C'(t)$ do pôvodnej nehomogénnej rovnice \eqref{eq:diff_num}:
$$ \underbrace{K'(t) \cdot e^{-\frac{t}{200}} - \frac{1}{200} K(t) \cdot e^{-\frac{t}{200}}}_{u_C'(t)} + \frac{1}{200} \cdot \underbrace{\left( K(t) \cdot e^{-\frac{t}{200}} \right)}_{u_C(t)} = \frac{28}{200} $$

Členy s $K(t)$ sa odčítajú:
$$ K'(t) \cdot e^{-\frac{t}{200}} = \frac{28}{200} $$

Vyjadríme $K'(t)$:
$$ K'(t) = \frac{28}{200} \cdot e^{\frac{t}{200}} $$

Integrujeme pre získanie $K(t)$ ($k$ je integračná konštanta):
$$ K(t) = \int \frac{28}{200} \cdot e^{\frac{t}{200}} \, \mathrm{d}t = \frac{28}{200} \cdot 200 \cdot e^{\frac{t}{200}} + k = 28 \cdot e^{\frac{t}{200}} + k $$

\textbf{C) Všeobecné riešenie:}
Dosadíme $K(t)$ naspäť do vzťahu pre $u_C(t)$:
$$ u_C(t) = \left( 28 \cdot e^{\frac{t}{200}} + k \right) \cdot e^{-\frac{t}{200}} $$
$$ u_C(t) = 28 \cdot e^{0} + k \cdot e^{-\frac{t}{200}} = 28 + k \cdot e^{-\frac{t}{200}} $$

\textbf{D) Výpočet konštanty $k$ z počiatočných podmienok:}
Pre $t=0$ platí $u_C(0) = \qty{5}{\volt}$.
$$ 5 = 28 + k \cdot e^{0} \implies k = 5 - 28 = -23 $$

\textbf{Výsledná rovnica napätia na kondenzátore:}
\begin{equation}
    u_C(t) = 28 - 23 \cdot e^{-\frac{t}{200}} \quad [\si{\volt}]
\end{equation}

\subsection*{3. Kontrola výpočtu (Skúška)}
Dosadíme výsledok do zostavenej diferenciálnej rovnice \eqref{eq:diff_num}.

Ľavá strana (L):
$$ u_C'(t) = -23 \cdot \left( -\frac{1}{200} \right) \cdot e^{-\frac{t}{200}} = \frac{23}{200} e^{-\frac{t}{200}} $$
$$ L = u_C'(t) + \frac{1}{200} u_C(t) = \frac{23}{200} e^{-\frac{t}{200}} + \frac{1}{200} \left( 28 - 23 e^{-\frac{t}{200}} \right) $$
$$ L = \frac{23}{200} e^{-\frac{t}{200}} + \frac{28}{200} - \frac{23}{200} e^{-\frac{t}{200}} = \frac{28}{200} = \frac{7}{50} $$

Pravá strana (P):
$$ P = \frac{U}{RC} = \frac{28}{200} = \frac{7}{50} $$

$$ L = P \quad $$
